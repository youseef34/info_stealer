import phonenumbers
from phonenumbers import carrier, geocoder, timezone

def analyze_phone(phone):
    """
    تحليل رقم الهاتف وإرجاع معلومات عنه.
    """
    result = {
        "input": phone,
        "valid": False,
        "country": None,
        "carrier": None,
        "timezone": None,
        "line_type": None
    }
    try:
        parsed = phonenumbers.parse(phone, None)
        if phonenumbers.is_valid_number(parsed):
            result["valid"] = True
            result["country"] = geocoder.description_for_number(parsed, "en")
            result["carrier"] = carrier.name_for_number(parsed, "en")
            tz = timezone.time_zones_for_number(parsed)
            result["timezone"] = list(tz)[0] if tz else None
            num_type = phonenumbers.number_type(parsed)
            if num_type == phonenumbers.PhoneNumberType.MOBILE:
                result["line_type"] = "Mobile"
            elif num_type == phonenumbers.PhoneNumberType.FIXED_LINE:
                result["line_type"] = "Fixed"
            else:
                result["line_type"] = "Other"
        else:
            result["error"] = "رقم غير صالح"
    except Exception as e:
        result["error"] = str(e)
    return result

def check_hibp(email):
    """
    التحقق من البريد الإلكتروني في قاعدة HaveIBeenPwned.
    """
    if not re.match(r"[^@]+@[^@]+\.[^@]+", email):
        return {"error": "بريد غير صالح"}
    try:
        sha1 = hashlib.sha1(email.encode()).hexdigest().upper()
        prefix, suffix = sha1[:5], sha1[5:]
        resp = requests.get(f"{Config.HIBP}{prefix}", timeout=Config.TIMEOUT)
        if resp.status_code == 200:
            for line in resp.text.splitlines():
                if line.startswith(suffix):
                    count = int(line.split(':')[1])
                    return {"breach_count": count, "status": "مخترق" if count > 0 else "نظيف"}
        return {"breach_count": 0, "status": "غير معروف"}
    except Exception as e:
        return {"error": str(e)}

import dns.resolver
import whois

def dns_enum(domain):
    """استعلام DNS لجميع الأنواع الشائعة."""
    records = {}
    for rec_type in ['A', 'AAAA', 'MX', 'NS', 'TXT', 'CNAME']:
        try:
            answers = dns.resolver.resolve(domain, rec_type)
            records[rec_type] = [str(r) for r in answers]
        except:
            records[rec_type] = []
    return records

def subdomain_bruteforce(domain, wordlist=None):
    """تعداد النطاقات الفرعية باستخدام قائمة كلمات."""
    if wordlist is None:
        wordlist = ['www', 'mail', 'ftp', 'admin', 'dev', 'test', 'api', 'blog', 'shop', 'vpn', 'webmail']
    found = []
    for sub in wordlist:
        fqdn = f"{sub}.{domain}"
        try:
            dns.resolver.resolve(fqdn, 'A')
            found.append(fqdn)
        except:
            pass
    return found

def whois_lookup(domain):
    """استعلام WHOIS عن النطاق."""
    try:
        w = whois.whois(domain)
        return {
            "registrar": w.registrar,
            "creation_date": str(w.creation_date),
            "expiration_date": str(w.expiration_date),
            "name_servers": w.name_servers
        }
    except:
        return {"error": "فشل في استعلام WHOIS"}

def geoip(ip):
    """تحديد الموقع الجغرافي من عنوان IP باستخدام ip-api.com."""
    if not ip:
        return {}
    try:
        url = f"{Config.IP_API}{ip}?fields=status,country,city,regionName,lat,lon,isp,org"
        resp = requests.get(url, timeout=Config.TIMEOUT)
        data = resp.json()
        if data.get('status') == 'success':
            return {
                "country": data.get('country'),
                "city": data.get('city'),
                "region": data.get('regionName'),
                "latitude": data.get('lat'),
                "longitude": data.get('lon'),
                "isp": data.get('isp'),
                "organization": data.get('org'),
                "google_maps": f"https://www.google.com/maps?q={data.get('lat')},{data.get('lon')}"
            }
        return {}
    except:
        return {}

def shodan_lookup(ip, api_key=None):
    """البحث في Shodan عن معلومات IP (تتطلب مفتاح API)."""
    if not api_key:
        return {"error": "مفتاح Shodan غير موجود"}
    try:
        import shodan
        api = shodan.Shodan(api_key)
        info = api.host(ip)
        return {
            "hostnames": info.get('hostnames'),
            "ports": info.get('ports'),
            "vulns": info.get('vulns'),
            "org": info.get('org')
        }
    except:
        return {"error": "فشل في Shodan"}

def abuseipdb_check(ip, api_key=None):
    """التحقق من سمعة IP عبر AbuseIPDB."""
    if not api_key:
        return {"error": "مفتاح AbuseIPDB غير موجود"}

        url = f"https://api.abuseipdb.com/api/v2/check?ipAddress={ip}"
        headers = {"Key": api_key, "Accept": "application/json"}

            data = resp.json()
            return {
                "abuse_score": data['data']['abuseConfidenceScore'],
                "country": data['data']['countryCode']
            }
        return {"error": "فشل في الاستعلام"}
    except:
        return {"error": "استثناء"}

    """إرسال طلب HTTP مع إعادة المحاولات ودعم الوكيل والتأخير العشوائي."""
